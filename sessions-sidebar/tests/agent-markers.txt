# Run only in a fresh isolated native window with sessions-sidebar linked and an active shell tab
# (any display mode). From the plugin folder:
# tern ctl --control <endpoint> --file tests/agent-markers.txt
# Needs no fixture files: the first step writes three hand-written TSP recordings to
# /tmp/sessions-sidebar-agent-{finish,rework,ask}.jsonl, played by absolute path (Tern 0.8.2
# resolves relative `feed` paths only under STENCIL_FIXTURE_ROOT).
# `surface-play … paced` turns the focused pane into a stand-in `omp` agent and keeps playing after
# focus moves, so its state changes while its tab is not active (`feed` writes only the focused
# pane, whose tab is then active). OSC 9;4;3 is working, 9;4;0 idle, 9;4;4;50 paused (waiting_input).
# finish: working, idle at 8 s. rework: working, idle at 8 s, working again at 16 s, idle at
# 24 s. ask: idle, waiting_input at 3 s.
# The sidebar's 1 s timer can run far slower in a hidden or loaded window, so every check driven
# by a recording first forces a refresh (`plugins run plugin.sessions-sidebar.refresh`), and each
# working phase is observed by such a refresh before its finish is checked. Waits are measured from
# the recording's start, so slow commands only make the idle checks later, never early.
# Every check also asserts that no row has an activity strip (role sessions-sidebar-status).
# Opens three tabs and one extra session; closes nothing.
ready
wait 1500
carly lua "agentTest = {}; local function recording(steps) local nl = string.char(10); local out = ''; for _, step in steps do out = out .. tern.json.encode({ t = step[1], dir = 'out', verb = 'osc', body = step[2] }) .. nl end; return out end; tern.fs.write('/tmp/sessions-sidebar-agent-finish.jsonl', recording({ { 0, '9;4;3' }, { 8000, '9;4;0' } })); tern.fs.write('/tmp/sessions-sidebar-agent-rework.jsonl', recording({ { 0, '9;4;3' }, { 8000, '9;4;0' }, { 16000, '9;4;3' }, { 24000, '9;4;0' } })); tern.fs.write('/tmp/sessions-sidebar-agent-ask.jsonl', recording({ { 0, '9;4;0' }, { 3000, '9;4;4;50' } })); function agentMarksOf(session) local tab = cx.session:tab_of(cx.session:focused()); local sidebar; for _, canvas in cx.canvas:list() do if canvas.owner == 'sessions-sidebar' and cx.session:tab_of(canvas.pane) == tab then sidebar = canvas.pane end end; assert(sidebar, 'no sidebar in the active tab'); local info = cx.canvas:get(sidebar); assert(info and info.view, 'sidebar has no view'); local rowKey = 'session-' .. string.format('%.0f', session); local found = {}; local function walk(node, inRow) local p = node.p or {}; local here = inRow or p.key == rowKey; if p.key == rowKey then found.row = true end; if p.role == 'sessions-sidebar-status' then found.status = true end; if here and p.role == 'sessions-sidebar-question' then found.question = p.text; found.questionTitle = p.title end; if here and p.role == 'sessions-sidebar-working' then found.working = node.k; found.workingStyle = p.style; found.workingTitle = p.title end; if here and p.role == 'sessions-sidebar-done' then found.done = p.text; found.doneTitle = p.title end; for _, child in node.c or {} do walk(child, here) end end; walk(info.view, false); assert(found.row, 'no sidebar row for session ' .. rowKey); return found end; function expectMarks(label, session, question, working, done) local m = agentMarksOf(session); assert(not m.status, label .. ': a row still has an activity strip'); assert((m.question ~= nil) == question, label .. ': question marker is ' .. tostring(m.question)); assert((m.working ~= nil) == working, label .. ': working spinner is ' .. tostring(m.working)); assert((m.done ~= nil) == done, label .. ': done marker is ' .. tostring(m.done)); assert(m.question == nil or m.question == '?', label .. ': question text ' .. tostring(m.question)); assert(m.working == nil or (m.working == 'spinner' and m.workingStyle == 'starburst'), label .. ': working node ' .. tostring(m.working) .. ' ' .. tostring(m.workingStyle)); assert(m.done == nil or m.done == utf8.char(0x2713), label .. ': done text ' .. tostring(m.done)); return m end; agentTest.home = cx.session:focused(); assert(agentTest.home, 'no content focus'); agentTest.homeTab = cx.session:tab_of(agentTest.home); agentTest.session = cx.sessions:current().id; agentTest.sessionName = cx.sessions:current().name; expectMarks('start', agentTest.session, false, false, false); return 'PASS setup: no markers without agents'"
carly ok
carly close
