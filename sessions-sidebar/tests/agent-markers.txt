# Run only in a fresh isolated native window with sessions-sidebar linked and an active shell tab
# (any display mode). From the plugin folder:
# tern ctl --control <endpoint> --file tests/agent-markers.txt
# Needs no fixture files: the first step writes two hand-written TSP recordings to
# /tmp/sessions-sidebar-agent-finish.jsonl and /tmp/sessions-sidebar-agent-ask.jsonl, played by
# absolute path (Tern 0.8.2 resolves relative `feed` paths only under STENCIL_FIXTURE_ROOT).
# `surface-play … paced` turns the focused pane into a stand-in `omp` agent and keeps playing after
# focus moves, so its state changes while its tab is not active (`feed` writes only the focused
# pane, whose tab is then active). finish: OSC 9;4;3 (working), 2.5 s later OSC 9;4;0 (idle);
# ask: idle, 2.5 s later OSC 9;4;4;50 (paused: waiting_input).
# Opens three tabs and one extra session; closes nothing.
ready
wait 1500
carly lua "agentTest = {}; local function recording(first, later) local nl = string.char(10); return tern.json.encode({ t = 0, dir = 'out', verb = 'osc', body = first }) .. nl .. tern.json.encode({ t = 2500, dir = 'out', verb = 'osc', body = later }) .. nl end; tern.fs.write('/tmp/sessions-sidebar-agent-finish.jsonl', recording('9;4;3', '9;4;0')); tern.fs.write('/tmp/sessions-sidebar-agent-ask.jsonl', recording('9;4;0', '9;4;4;50')); function agentMarksOf(session) local tab = cx.session:tab_of(cx.session:focused()); local sidebar; for _, canvas in cx.canvas:list() do if canvas.owner == 'sessions-sidebar' and cx.session:tab_of(canvas.pane) == tab then sidebar = canvas.pane end end; assert(sidebar, 'no sidebar in the active tab'); local info = cx.canvas:get(sidebar); assert(info and info.view, 'sidebar has no view'); local rowKey = 'session-' .. string.format('%.0f', session); local found = {}; local function walk(node, inRow) local p = node.p or {}; local here = inRow or p.key == rowKey; if p.key == rowKey then found.row = true end; if here and p.role == 'sessions-sidebar-question' then found.question = p.text; found.questionTitle = p.title end; if here and p.role == 'sessions-sidebar-done' then found.done = p.text; found.doneTitle = p.title end; for _, child in node.c or {} do walk(child, here) end end; walk(info.view, false); assert(found.row, 'no sidebar row for session ' .. rowKey); return found end; function expectMarks(label, session, question, done) local m = agentMarksOf(session); assert((m.question ~= nil) == question, label .. ': question marker is ' .. tostring(m.question)); assert((m.done ~= nil) == done, label .. ': done marker is ' .. tostring(m.done)); assert(m.question == nil or m.question == '?', label .. ': question text ' .. tostring(m.question)); assert(m.done == nil or m.done == utf8.char(0x2713), label .. ': done text ' .. tostring(m.done)); return m end; agentTest.home = cx.session:focused(); assert(agentTest.home, 'no content focus'); agentTest.homeTab = cx.session:tab_of(agentTest.home); agentTest.session = cx.sessions:current().id; agentTest.sessionName = cx.sessions:current().name; expectMarks('start', agentTest.session, false, false); return 'PASS setup: no markers without agents'"
carly ok
